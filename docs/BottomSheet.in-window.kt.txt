/*
 * 바텀 시트.
 *
 * Material ModalBottomSheet 은 별도 창(다이얼로그)에 그려져서, 닫힐 때 창 포커스가 돌아오며
 * 일부 기기(One UI 8.5 / Android 16 확인)에서 키보드가 잠깐 떴다 내려간다.
 * 그래서 별도 창 없이 같은 창 안([BottomSheetHost])에 직접 그린다.
 */

/** 화면 위에 시트를 그리는 자리. 앱 루트(MainActivity)에 한 번 둔다. */
@Composable
fun BottomSheetHost(content: @Composable () -> Unit) {
    val state: BottomSheetHostState = remember { BottomSheetHostState() }
    val entry: BottomSheetEntry? = state.entry

    CompositionLocalProvider(LocalBottomSheetHost provides state) {
        Box(modifier = Modifier.fillMaxSize()) {
            // 시트가 열린 동안 뒤 화면은 낭독기에서 숨긴다 (모달)
            Box(modifier = if (entry != null) Modifier.clearAndSetSemantics {} else Modifier) {
                content()
            }
            if (entry != null) {
                key(entry.id) {
                    SheetOverlay(entry = entry)
                }
            }
        }
    }
}

/**
 * 화면에서 시트를 선언하고, 반환된 손잡이로 연다. 화면마다 `if` 나 열림 상태를 두지 않아도 된다.
 *
 * ```
 * val sortSheet: BottomSheetHandle = rememberBottomSheet { dismiss -> ... }
 * SortChip(onClick = { sortSheet.show() })
 * ```
 *
 * - 열림 여부는 rememberSaveable 로 저장되어 화면 회전, 프로세스 종료 후에도 복원된다.
 * - [content] 는 매번 다시 그려지므로 열려 있는 동안 바뀐 상태도 바로 반영된다.
 * - ViewModel 이 열 시점을 정해야 하면 Effect 를 받아 `show()` 를 부른다.
 *
 * @param onDismiss 시트가 닫힌 뒤 호출된다. (dismiss 호출, 뒷배경 터치, 끌어내리기, 뒤로가기 모두)
 * @param content 시트 내용. 받은 `dismiss` 를 부르면 닫힘 애니메이션 후 닫힌다.
 */
@Composable
fun rememberBottomSheet(
    onDismiss: () -> Unit = {},
    content: @Composable ColumnScope.(dismiss: () -> Unit) -> Unit
): BottomSheetHandle {
    val isShown: MutableState<Boolean> = rememberSaveable { mutableStateOf(false) }
    val handle: BottomSheetHandle = remember(isShown) { BottomSheetHandle(isShown) }

    if (isShown.value) {
        BottomSheet(
            onDismissRequest = {
                isShown.value = false
                onDismiss()
            },
            content = content
        )
    }
    return handle
}

/** [rememberBottomSheet] 가 돌려주는 손잡이 */
@Stable
class BottomSheetHandle internal constructor(private val isShown: MutableState<Boolean>) {
    fun show() {
        isShown.value = true
    }
}

/**
 * 시트를 [BottomSheetHost] 에 띄운다. 화면에서는 보통 [rememberBottomSheet] 를 쓴다.
 * 이 함수가 composition 에 있는 동안 시트가 열려 있다.
 */
@Composable
fun BottomSheet(onDismissRequest: () -> Unit, content: @Composable ColumnScope.(dismiss: () -> Unit) -> Unit) {
    val host: BottomSheetHostState = LocalBottomSheetHost.current
    val currentOnDismissRequest: () -> Unit by rememberUpdatedState(onDismissRequest)
    val currentContent: @Composable ColumnScope.(dismiss: () -> Unit) -> Unit by rememberUpdatedState(content)
    val entry: BottomSheetEntry = remember {
        BottomSheetEntry(
            id = Any(),
            onDismissRequest = { currentOnDismissRequest() },
            content = { dismiss -> currentContent(dismiss) }
        )
    }

    DisposableEffect(host, entry) {
        host.show(entry)
        onDispose { host.remove(entry.id) }
    }
}

@Stable
internal class BottomSheetHostState {
    var entry: BottomSheetEntry? by mutableStateOf(null)
        private set

    fun show(entry: BottomSheetEntry) {
        this.entry = entry
    }

    fun remove(id: Any) {
        if (entry?.id === id) entry = null
    }
}

internal class BottomSheetEntry(
    val id: Any,
    val onDismissRequest: () -> Unit,
    val content: @Composable ColumnScope.(dismiss: () -> Unit) -> Unit
)

private val LocalBottomSheetHost: ProvidableCompositionLocal<BottomSheetHostState> =
    staticCompositionLocalOf { error("BottomSheet 은 BottomSheetHost 안에서만 쓸 수 있습니다. 앱 루트에 BottomSheetHost 를 두세요.") }

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun SheetOverlay(entry: BottomSheetEntry) {
    val colors: MbColors = MyBooksTheme.colors
    val keyboardController: SoftwareKeyboardController? = LocalSoftwareKeyboardController.current
    // 열 때 키보드가 올라와 있었으면 시트를 가리지 않게 내리고, 닫은 뒤 다시 올린다
    val isImeVisible: Boolean = WindowInsets.isImeVisible
    val restoreKeyboard: Boolean = remember { isImeVisible }
    val visibleState: MutableTransitionState<Boolean> = remember { MutableTransitionState(false) }
    var isDismissing: Boolean by remember { mutableStateOf(false) }
    var dragOffset: Float by remember { mutableFloatStateOf(0f) }
    var sheetHeight: Int by remember { mutableIntStateOf(0) }
    val closeLabel: String = stringResource(R.string.core_designsystem_sheet_close)
    val dismiss: () -> Unit = {
        if (!isDismissing) {
            isDismissing = true
            visibleState.targetState = false
        }
    }

    LaunchedEffect(Unit) {
        if (restoreKeyboard) keyboardController?.hide()
        visibleState.targetState = true
    }
    // 닫힘 애니메이션이 끝나면 정리
    LaunchedEffect(Unit) {
        snapshotFlow { isDismissing && visibleState.isIdle && !visibleState.currentState }.first { it }
        if (restoreKeyboard) keyboardController?.show()
        entry.onDismissRequest()
    }
    BackHandler(enabled = !isDismissing, onBack = dismiss)

    AnimatedVisibility(
        visibleState = visibleState,
        enter = EnterTransition.None,
        exit = ExitTransition.None
    ) {
        Box(modifier = Modifier.fillMaxSize()) {
            // 뒷배경: 터치하면 닫힘
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .animateEnterExit(enter = fadeIn(), exit = fadeOut())
                    .background(colors.scrim)
                    .pointerInput(Unit) { detectTapGestures { dismiss() } }
                    .semantics {
                        contentDescription = closeLabel
                        onClick {
                            dismiss()
                            true
                        }
                    }
            )
            // 시트: 아래로 끌어서 닫을 수 있다
            Column(
                modifier = Modifier
                    .align(Alignment.BottomCenter)
                    .animateEnterExit(
                        enter = slideInVertically { fullHeight -> fullHeight },
                        exit = slideOutVertically { fullHeight -> fullHeight }
                    )
                    .offset { IntOffset(x = 0, y = dragOffset.roundToInt()) }
                    .onSizeChanged { size -> sheetHeight = size.height }
                    .fillMaxWidth()
                    .clip(RoundedCornerShape(topStart = 20.dp, topEnd = 20.dp))
                    .background(colors.card)
                    // 시트 안의 빈 곳을 눌러도 뒷배경으로 전달되지 않게 한다
                    .pointerInput(Unit) { detectTapGestures { } }
                    .draggable(
                        orientation = Orientation.Vertical,
                        state = rememberDraggableState { delta -> dragOffset = (dragOffset + delta).coerceAtLeast(0f) },
                        onDragStopped = { velocity ->
                            if (dragOffset > sheetHeight * 0.3f || velocity > 1500f) {
                                dismiss()
                            } else {
                                animate(initialValue = dragOffset, targetValue = 0f) { value, _ -> dragOffset = value }
                            }
                        }
                    )
                    .windowInsetsPadding(WindowInsets.navigationBars)
                    .semantics { isTraversalGroup = true }
            ) {
                Handle(modifier = Modifier.align(Alignment.CenterHorizontally))
                entry.content(this, dismiss)
            }
        }
    }
}

@Composable
private fun Handle(modifier: Modifier = Modifier) {
    Box(
        modifier = modifier
            .padding(vertical = 12.dp)
            .size(width = 40.dp, height = 4.dp)
            .clip(RoundedCornerShape(2.dp))
            .background(MyBooksTheme.colors.border)
    )
}
